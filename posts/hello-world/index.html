package user

import (
    "fmt"
    "net/mail"
    "uuid"
)

type User struct {
    id    uuid.UUID
    email string
}

func NewUser(email string) (u User, err error) {
    _, err = mail.ParseAddress(email)
    if err != nil {
        err = fmt.Errorf("invalid email %q: %w", email, err)
        return
    }
    u.id = uuid.NewV7()
    u.email = email
    return
}

func (u *User) ID() uuid.UUID { return u.id }
func (u *User) Email() string { return u.email }
